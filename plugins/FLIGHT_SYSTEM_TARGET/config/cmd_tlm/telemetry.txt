  ITEM RTC_TIMESTAMP 16 64 FLOAT "Timestamp (Unix seconds)"
    FORMAT_STRING "%.3f"

  # Sensor Data (all floats, 4 bytes = 32 bits each)
  # Offsets are relative to the start of the packet (after PKT_ID)
  # PKT_ID is 0-15 bits. RTC_TIMESTAMP is 16-79 bits.
  # First sensor data item starts at bit 80.
